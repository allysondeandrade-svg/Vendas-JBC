Onde ficam salvas as vendas?

Ficam salvas no Armazenamento Local Persistente do Navegador/Celular (WebStorage / IndexedDB). Isso significa que:

  • ✅ 100% Offline: Você pode registrar vendas sem precisar de conexão com a internet.
  • ✅ Privacidade Total: Seus dados não vão para servidores de terceiros.
  • ✅ Não Apaga: Ficam guardados mesmo fechando o navegador ou desligando o celular.
  • ⚠️ Multi-aparelhos: Como os dados ficam no dispositivo, para usar em vários celulares simultâneos você pode sincronizar via Planilha Google ou Nuvem.

/**
 * GOOGLE APPS SCRIPT - API BANCO DE DADOS 24H
 * Registro de Vendas & Catálogo de Produtos
 */
function doGet(e) {
  var action = (e && e.parameter && e.parameter.action) ? e.parameter.action : "getAll";
  if (action === "ping") {
    return createJsonResponse({ status: "success", message: "Conexão ativa 24h!" });
  }
  var allProds = readProductsFromSheet();
  var allSales = readSalesFromSheet();
  return createJsonResponse({ status: "success", products: allProds, sales: allSales });
}

function doPost(e) {
  try {
    if (!e || !e.postData || !e.postData.contents) {
      return createJsonResponse({ status: "error", message: "Dados não enviados" });
    }
    var payload = JSON.parse(e.postData.contents);
    var action = payload.action;

    if (action === "ping") {
      return createJsonResponse({ status: "success", message: "Conexão com Google Sheets ativa!" });
    }

    if (action === "addSale") {
      var sale = payload.sale || payload;
      var sheet = getOrCreateSheet("Vendas", [
        "Data/Hora", "Vendedor", "Produto", "Preço (R$)", 
        "Cliente", "Contato", "Endereço", "Frete", 
        "Pagamento", "Observação", "Comissão (R$)", "Lucro (R$)", "ID Venda"
      ]);
      
      // Se houver múltiplos itens anotados, gera uma linha para cada produto
      if (sale.itens && Array.isArray(sale.itens) && sale.itens.length > 0) {
        var count = sale.itens.length;
        for (var idx = 0; idx < count; idx++) {
          var item = sale.itens[idx];
          var itemNome = item.produto || sale.produto || "";
          var itemPreco = (typeof item.preco === "number") ? item.preco.toLocaleString("pt-BR", { minimumFractionDigits: 2, maximumFractionDigits: 2 }) : (item.preco || "0,00");
          
          // Calcula comissão e lucro do item (ou rateia se não houver individual)
          var itemComissao = "0,00";
          var itemLucro = "0,00";
          if (typeof item.basePrice === "number" && typeof item.preco === "number") {
            var cVal = item.preco - item.basePrice;
            itemComissao = cVal.toLocaleString("pt-BR", { minimumFractionDigits: 2, maximumFractionDigits: 2 });
          }
          if (typeof item.basePrice === "number" && typeof item.cost === "number") {
            var lVal = item.basePrice - item.cost;
            itemLucro = lVal.toLocaleString("pt-BR", { minimumFractionDigits: 2, maximumFractionDigits: 2 });
          }

          sheet.appendRow([
            sale.dateString || new Date().toLocaleString("pt-BR"),
            sale.vendedor || "",
            itemNome,
            itemPreco,
            sale.cliente || "",
            sale.contato || "",
            sale.endereco || "",
            (idx === 0) ? (sale.frete || "") : "Incluso",
            sale.pagamento || "",
            (count > 1 ? ("Item " + (idx + 1) + "/" + count + " - " + (sale.observacao || "")) : (sale.observacao || "")),
            itemComissao !== "0,00" ? itemComissao : (sale.comissao || "0,00"),
            itemLucro !== "0,00" ? itemLucro : (sale.lucro || "0,00"),
            (sale.id || ("sale-" + new Date().getTime())) + (count > 1 ? ("-" + (idx + 1)) : "")
          ]);
        }
      } else {
        sheet.appendRow([
          sale.dateString || new Date().toLocaleString("pt-BR"),
          sale.vendedor || "", sale.produto || "", sale.preco || "0,00",
          sale.cliente || "", sale.contato || "", sale.endereco || "",
          sale.frete || "", sale.pagamento || "", sale.observacao || "",
          sale.comissao || "0,00", sale.lucro || "0,00",
          sale.id || ("sale-" + new Date().getTime())
        ]);
      }
      return createJsonResponse({ status: "success", message: "Venda gravada com sucesso!" });
    }

    if (action === "editSale") {
      var sale = payload.sale;
      var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("Vendas");
      if (sheet) {
        var data = sheet.getDataRange().getValues();
        for (var i = 1; i < data.length; i++) {
          if (String(data[i][12]) === String(sale.id)) {
            sheet.getRange(i + 1, 1, 1, 13).setValues([[
              sale.dateString || String(data[i][0]),
              sale.vendedor || "", sale.produto || "", sale.preco || "0,00",
              sale.cliente || "", sale.contato || "", sale.endereco || "",
              sale.frete || "", sale.pagamento || "", sale.observacao || "",
              sale.comissao || "0,00", sale.lucro || "0,00", sale.id
            ]]);
            return createJsonResponse({ status: "success", message: "Venda editada!" });
          }
        }
      }
      return createJsonResponse({ status: "error", message: "Venda não encontrada" });
    }

    if (action === "deleteSale") {
      var id = payload.id;
      var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("Vendas");
      if (sheet) {
        var data = sheet.getDataRange().getValues();
        for (var i = 1; i < data.length; i++) {
          if (String(data[i][12]) === String(id)) {
            sheet.deleteRow(i + 1);
            return createJsonResponse({ status: "success", message: "Venda excluída!" });
          }
        }
      }
      return createJsonResponse({ status: "error", message: "Venda não encontrada" });
    }

    if (action === "saveProducts") {
      var products = payload.products || [];
      var sheet = getOrCreateSheet("Produtos", [
        "SKU", "Produto", "Preço Venda (R$)", "Custo (R$)", "Preço Base (R$)", "Categoria", "ID"
      ]);
      var lastRow = sheet.getLastRow();
      if (lastRow > 1) {
        sheet.getRange(2, 1, lastRow - 1, 7).clearContent();
      }
      if (products.length > 0) {
        var rows = products.map(function(p) {
          return [
            p.sku || p.id || "",
            p.name || "",
            p.price || "0,00",
            p.cost || "",
            p.basePrice || p.price || "0,00",
            p.category || "Geral",
            p.id || ("p-" + Math.random().toString(36).substr(2, 9))
          ];
        });
        sheet.getRange(2, 1, rows.length, 7).setValues(rows);
      }
      return createJsonResponse({ status: "success", message: products.length + " produtos salvos na planilha!", count: products.length });
    }

    return createJsonResponse({ status: "error", message: "Ação não reconhecida" });
  } catch (error) {
    return createJsonResponse({ status: "error", message: error.toString() });
  }
}

function getOrCreateSheet(sheetName, headers) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName(sheetName);
  if (!sheet) {
    sheet = ss.insertSheet(sheetName);
    sheet.appendRow(headers);
    sheet.getRange(1, 1, 1, headers.length).setFontWeight("bold").setBackground("#e2e8f0");
    sheet.setFrozenRows(1);
  }
  return sheet;
}

function readProductsFromSheet() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName("Produtos");
  if (!sheet) return [];
  var data = sheet.getDataRange().getValues();
  if (data.length <= 1) return [];
  var products = [];
  for (var i = 1; i < data.length; i++) {
    var row = data[i];
    if (!row[1]) continue;
    products.push({
      sku: String(row[0] || ""),
      name: String(row[1]),
      price: String(row[2]),
      cost: String(row[3] || ""),
      basePrice: String(row[4] || row[2]),
      category: String(row[5] || "Geral"),
      id: String(row[6] || ("p-" + i))
    });
  }
  return products;
}

function readSalesFromSheet() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName("Vendas");
  if (!sheet) return [];
  var data = sheet.getDataRange().getValues();
  if (data.length <= 1) return [];
  var sales = [];
  for (var i = 1; i < data.length; i++) {
    var row = data[i];
    if (!row[1] && !row[2]) continue;
    sales.push({
      id: String(row[12] || ("sale-" + i)),
      dateString: String(row[0]),
      vendedor: String(row[1]),
      produto: String(row[2]),
      preco: String(row[3]),
      cliente: String(row[4]),
      contato: String(row[5]),
      endereco: String(row[6]),
      frete: String(row[7]),
      pagamento: String(row[8]),
      observacao: String(row[9]),
      comissao: String(row[10]),
      lucro: String(row[11]),
      timestamp: new Date(row[0]).getTime() || (Date.now() - i * 1000)
    });
  }
  return sales;
}

function createJsonResponse(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON);
}
